```sql
-- GC 삭제 로그 테이블 생성
CREATE TABLE IF NOT EXISTS gc_deletion_log (
    id BIGSERIAL PRIMARY KEY,
    annotation_id INTEGER NOT NULL,
    snapshot_image_key TEXT NOT NULL,
    file_size BIGINT,
    deleted_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    dry_run BOOLEAN NOT NULL DEFAULT FALSE,
    status TEXT NOT NULL CHECK (status IN ('success', 'failed', 'skipped')),
    error_message TEXT,
    
    -- 인덱스
    CONSTRAINT fk_annotation FOREIGN KEY (annotation_id) 
        REFERENCES annotation_annotation(id) ON DELETE CASCADE
);

-- 인덱스 생성
CREATE INDEX idx_gc_deletion_log_annotation_id ON gc_deletion_log(annotation_id);
CREATE INDEX idx_gc_deletion_log_deleted_at ON gc_deletion_log(deleted_at);
CREATE INDEX idx_gc_deletion_log_status ON gc_deletion_log(status);

-- 코멘트 추가
COMMENT ON TABLE gc_deletion_log IS 'GC 작업으로 삭제된 스냅샷 이미지 로그';
COMMENT ON COLUMN gc_deletion_log.annotation_id IS '삭제된 스냅샷의 어노테이션 ID';
COMMENT ON COLUMN gc_deletion_log.snapshot_image_key IS '삭제된 S3 object key';
COMMENT ON COLUMN gc_deletion_log.file_size IS '삭제된 파일 크기 (바이트)';
COMMENT ON COLUMN gc_deletion_log.deleted_at IS '삭제 시각';
COMMENT ON COLUMN gc_deletion_log.dry_run IS 'Dry-run 모드 여부';
COMMENT ON COLUMN gc_deletion_log.status IS '삭제 상태 (success, failed, skipped)';
COMMENT ON COLUMN gc_deletion_log.error_message IS '실패 시 에러 메시지';
```

### 3.2 GcRepository 구현

**경로**: `pacs-server/src/infrastructure/repositories/gc_repository_impl.rs`

```rust
use async_trait::async_trait;
use sqlx::PgPool;
use std::sync::Arc;
use crate::domain::entities::annotation::{Annotation, SnapshotUploadStatus};
use crate::domain::repositories::GcRepository;

pub struct GcRepositoryImpl {
    pool: Arc<PgPool>,
}

impl GcRepositoryImpl {
    pub fn new(pool: Arc<PgPool>) -> Self {
        Self { pool }
    }
}

#[async_trait]
impl GcRepository for GcRepositoryImpl {
    async fn find_pending_snapshots(
        &self,
        grace_days: i32,
        batch_size: i32,
    ) -> Result<Vec<Annotation>, sqlx::Error> {
        sqlx::query_as::<_, Annotation>(
            r#"
            SELECT * FROM annotation_annotation
            WHERE snapshot_status = $1
              AND updated_at < NOW() - INTERVAL '1 day' * $2
            ORDER BY updated_at ASC
            LIMIT $3
            "#
        )
        .bind(SnapshotUploadStatus::Pending)
        .bind(grace_days)
        .bind(batch_size)
        .fetch_all(self.pool.as_ref())
        .await
    }

    async fn find_failed_snapshots(
        &self,
        grace_days: i32,
        batch_size: i32,
    ) -> Result<Vec<Annotation>, sqlx::Error> {
        sqlx::query_as::<_, Annotation>(
            r#"
            SELECT * FROM annotation_annotation
            WHERE snapshot_status = $1
              AND snapshot_image_key IS NOT NULL
              AND updated_at < NOW() - INTERVAL '1 day' * $2
            ORDER BY updated_at ASC
            LIMIT $3
            "#
        )
        .bind(SnapshotUploadStatus::Failed)
        .bind(grace_days)
        .bind(batch_size)
        .fetch_all(self.pool.as_ref())
        .await
    }

    async fn update_snapshot_status(
        &self,
        annotation_id: i32,
        status: SnapshotUploadStatus,
    ) -> Result<(), sqlx::Error> {
        sqlx::query(
            r#"
            UPDATE annotation_annotation
            SET snapshot_status = $1,
                updated_at = NOW()
            WHERE id = $2
            "#
        )
        .bind(status)
        .bind(annotation_id)
        .execute(self.pool.as_ref())
        .await?;

        Ok(())
    }
}
```

```rust
use sqlx::PgPool;

use crate::domain::entities::gc_deletion_log::{GcDeletionLog, NewGcDeletionLog};

                annotation_id, snapshot_image_key, file_size,
                dry_run, status, error_message
            )
            VALUES ($1, $2, $3, $4, $5, $6)

            "#

        .bind(log.snapshot_image_key)

        .bind(log.dry_run)

            SELECT * FROM gc_deletion_log

            "#

        .bind(end_date)
        .fetch_all(self.pool.as_ref())

    }
```

### 4.2 GcService 구현 (Part 1)

**경로**: `pacs-server/src/application/services/gc_service_impl.rs`

```rust
use async_trait::async_trait;
use std::sync::Arc;
use crate::application::services::{GcService, GcResult, ObjectStorageService};
use crate::domain::entities::annotation::SnapshotUploadStatus;
use crate::domain::entities::gc_deletion_log::NewGcDeletionLog;
use crate::domain::repositories::{GcRepository, GcLogRepository};
use crate::domain::ServiceError;

pub struct GcServiceImpl {
    gc_repository: Arc<dyn GcRepository>,
    gc_log_repository: Arc<dyn GcLogRepository>,
    object_storage: Arc<dyn ObjectStorageService>,
}

impl GcServiceImpl {
    pub fn new(
        gc_repository: Arc<dyn GcRepository>,
        gc_log_repository: Arc<dyn GcLogRepository>,
        object_storage: Arc<dyn ObjectStorageService>,
    ) -> Self {
        Self {
            gc_repository,
            gc_log_repository,
            object_storage,
        }
    }
}

#[async_trait]
impl GcService for GcServiceImpl {
    async fn timeout_pending_snapshots(
        &self,
        grace_days: i32,
        batch_size: i32,
        dry_run: bool,
    ) -> Result<Vec<GcResult>, ServiceError> {
        // 1. PENDING 상태 조회
        let annotations = self.gc_repository
            .find_pending_snapshots(grace_days, batch_size)
            .await?;

        let mut results = Vec::new();

        // 2. 각 어노테이션 처리
        for annotation in annotations {
            let snapshot_key = annotation.snapshot_image_key
                .clone()
                .unwrap_or_default();

            if dry_run {
                // Dry-run: 로그만 기록
                self.gc_log_repository.insert(NewGcDeletionLog {
                    annotation_id: annotation.id,
                    snapshot_image_key: snapshot_key.clone(),
                    file_size: None,
                    dry_run: true,
                    status: "skipped".to_string(),
                    error_message: None,
                }).await?;

                results.push(GcResult {
                    annotation_id: annotation.id,
                    snapshot_image_key: snapshot_key,
                    success: true,
                    error_message: None,
                });
            } else {
                // 실제 실행: 상태를 FAILED로 변경
                match self.gc_repository
                    .update_snapshot_status(annotation.id, SnapshotUploadStatus::Failed)
                    .await
                {
                    Ok(_) => {
                        // 성공 로그
                        self.gc_log_repository.insert(NewGcDeletionLog {
                            annotation_id: annotation.id,
                            snapshot_image_key: snapshot_key.clone(),
                            file_size: None,
                            dry_run: false,
                            status: "success".to_string(),
                            error_message: None,
                        }).await?;

                        results.push(GcResult {
                            annotation_id: annotation.id,
                            snapshot_image_key: snapshot_key,
                            success: true,
                            error_message: None,
                        });
                    }
                    Err(e) => {
                        // 실패 로그
                        let error_msg = e.to_string();
                        self.gc_log_repository.insert(NewGcDeletionLog {
                            annotation_id: annotation.id,
                            snapshot_image_key: snapshot_key.clone(),
                            file_size: None,
                            dry_run: false,
                            status: "failed".to_string(),
                            error_message: Some(error_msg.clone()),
                        }).await?;

                        results.push(GcResult {
                            annotation_id: annotation.id,
                            snapshot_image_key: snapshot_key,
                            success: false,
                            error_message: Some(error_msg),
                        });
                    }
                }
            }
        }

        Ok(results)
    }

    // cleanup_failed_snapshots() 구현은 다음 섹션에서...
}
```

### 4.3 GcService 구현 (Part 2 - cleanup_failed_snapshots)

**계속**: `pacs-server/src/application/services/gc_service_impl.rs`

```rust
#[async_trait]
impl GcService for GcServiceImpl {
    // ... timeout_pending_snapshots() 위에 구현됨 ...

    async fn cleanup_failed_snapshots(
        &self,
        grace_days: i32,
        batch_size: i32,
        dry_run: bool,
    ) -> Result<Vec<GcResult>, ServiceError> {
        // 1. FAILED 상태 조회
        let annotations = self.gc_repository
            .find_failed_snapshots(grace_days, batch_size)
            .await?;

        let mut results = Vec::new();

        // 2. 각 어노테이션 처리
        for annotation in annotations {
            let snapshot_key = match &annotation.snapshot_image_key {
                Some(key) => key.clone(),
                None => continue,  // snapshot_image_key가 없으면 스킵
            };

            if dry_run {
                // Dry-run: 로그만 기록
                self.gc_log_repository.insert(NewGcDeletionLog {
                    annotation_id: annotation.id,
                    snapshot_image_key: snapshot_key.clone(),
                    file_size: None,
                    dry_run: true,
                    status: "skipped".to_string(),
                    error_message: None,
                }).await?;

                results.push(GcResult {
                    annotation_id: annotation.id,
                    snapshot_image_key: snapshot_key,
                    success: true,
                    error_message: None,
                });
            } else {
                // 실제 실행: S3에서 삭제
                match self.object_storage.delete_file(&snapshot_key).await {
                    Ok(_) => {
                        // 성공 로그
                        self.gc_log_repository.insert(NewGcDeletionLog {
                            annotation_id: annotation.id,
                            snapshot_image_key: snapshot_key.clone(),
                            file_size: None,  // TODO: S3에서 파일 크기 조회 가능하면 추가
                            dry_run: false,
                            status: "success".to_string(),
                            error_message: None,
                        }).await?;

                        results.push(GcResult {
                            annotation_id: annotation.id,
                            snapshot_image_key: snapshot_key,
                            success: true,
                            error_message: None,
                        });
                    }
                    Err(e) => {
                        // 실패 로그
                        let error_msg = format!("S3 delete failed: {}", e);
                        self.gc_log_repository.insert(NewGcDeletionLog {
                            annotation_id: annotation.id,
                            snapshot_image_key: snapshot_key.clone(),
                            file_size: None,
                            dry_run: false,
                            status: "failed".to_string(),
                            error_message: Some(error_msg.clone()),
                        }).await?;

                        results.push(GcResult {
                            annotation_id: annotation.id,
                            snapshot_image_key: snapshot_key,
                            success: false,
                            error_message: Some(error_msg),
                        });
                    }
                }
            }
        }

        Ok(results)
    }
}
```

**경로**: `pacs-server/src/infrastructure/external/s3_object_storage_service.rs`

```rust
use async_trait::async_trait;
use aws_sdk_s3::Client as S3Client;
use crate::application::services::object_storage_service::{
    ObjectStorageError, ObjectStorageService, UploadedFile,
};

pub struct S3ObjectStorageService {
    client: S3Client,
    bucket_name: String,
    region: String,
}

impl S3ObjectStorageService {
    pub async fn new(
        bucket_name: String,
        region: String,
        access_key: String,
        secret_key: String,
    ) -> Result<Self, ObjectStorageError> {
        let config = aws_config::defaults(aws_config::BehaviorVersion::latest())
            .region(aws_config::Region::new(region.clone()))
            .credentials_provider(aws_sdk_s3::config::Credentials::new(
                access_key,
                secret_key,
                None,
                None,
                "pacs-server",
            ))
            .load()
            .await;

        let client = S3Client::new(&config);

        Ok(S3ObjectStorageService {
            client,
            bucket_name,
            region,
        })
    }

    fn generate_object_key(&self, file_path: &str) -> String {
        // snapshot_image_key는 이미 전체 경로를 포함하고 있음
        file_path.to_string()
    }
}

#[async_trait]
impl ObjectStorageService for S3ObjectStorageService {
    /// 파일 삭제 (GC Job 핵심 기능)
    async fn delete_file(&self, file_path: &str) -> Result<(), ObjectStorageError> {
        let object_key = self.generate_object_key(file_path);

        self.client
            .delete_object()
            .bucket(&self.bucket_name)
            .key(&object_key)
            .send()
            .await
            .map_err(|e| ObjectStorageError::S3Error(e.to_string()))?;

        Ok(())
    }

    /// 파일 메타데이터 조회 (파일 크기 확인)
    async fn get_file_metadata(&self, file_path: &str) -> Result<UploadedFile, ObjectStorageError> {
        let object_key = self.generate_object_key(file_path);

        let result = self
            .client
            .head_object()
            .bucket(&self.bucket_name)
            .key(&object_key)
            .send()
            .await
            .map_err(|e| ObjectStorageError::S3Error(e.to_string()))?;

        Ok(UploadedFile {
            file_path: file_path.to_string(),
            file_size: result.content_length().unwrap_or(0),
            checksum: result.e_tag().map(|s| s.to_string()),
            mime_type: result.content_type().map(|s| s.to_string()),
            last_modified: result.last_modified().map(|dt| dt.to_string()),
        })
    }

    /// 파일 존재 여부 확인
    async fn file_exists(&self, file_path: &str) -> Result<bool, ObjectStorageError> {
        let object_key = self.generate_object_key(file_path);

        match self
            .client
            .head_object()
            .bucket(&self.bucket_name)
            .key(&object_key)
            .send()
            .await
        {
            Ok(_) => Ok(true),
            Err(e) => {
                if e.to_string().contains("NoSuchKey") || e.to_string().contains("NotFound") {
                    Ok(false)
                } else {
                    Err(ObjectStorageError::S3Error(e.to_string()))
                }
            }
        }
    }
}
```

```rust
use clap::{Parser, Subcommand};
use pacs_server::application::services::{GcService, GcServiceImpl, ObjectStorageServiceFactory};
use pacs_server::infrastructure::repositories::{GcRepositoryImpl, GcLogRepositoryImpl};
use sqlx::postgres::PgPoolOptions;
use std::sync::Arc;

#[derive(Parser)]
#[command(name = "gc_runner")]
#[command(about = "PACS GC Batch Job Runner", long_about = None)]
struct Cli {
    #[command(subcommand)]
    command: Commands,
}

#[derive(Subcommand)]
enum Commands {
    /// Job A: PENDING 상태 타임아웃 처리
    TimeoutPending {
        /// Grace period (days)
        #[arg(long, default_value = "3")]
        grace_days: i32,

        /// Batch size
        #[arg(long, default_value = "1000")]
        batch_size: i32,

        /// Dry-run mode
        #[arg(long, default_value = "false")]
        dry_run: bool,
    },

    /// Job B: FAILED 상태 스냅샷 정리
    CleanupFailed {
        /// Grace period (days)
        #[arg(long, default_value = "7")]
        grace_days: i32,

        /// Batch size
        #[arg(long, default_value = "1000")]
        batch_size: i32,

        /// Dry-run mode
        #[arg(long, default_value = "false")]
        dry_run: bool,
    },
}

#[tokio::main]
async fn main() -> Result<(), Box<dyn std::error::Error>> {
    // 로깅 초기화
    env_logger::init();

    // CLI 파싱
    let cli = Cli::parse();

    // 환경 변수 로드
    let database_url = std::env::var("DATABASE_URL")
        .expect("DATABASE_URL must be set");
    let s3_bucket = std::env::var("S3_BUCKET")
        .expect("S3_BUCKET must be set");
    let s3_region = std::env::var("S3_REGION")
        .unwrap_or_else(|_| "us-east-1".to_string());
    let s3_access_key = std::env::var("S3_ACCESS_KEY")
        .expect("S3_ACCESS_KEY must be set");
    let s3_secret_key = std::env::var("S3_SECRET_KEY")
        .expect("S3_SECRET_KEY must be set");

    // DB 연결 풀 생성
    let pool = PgPoolOptions::new()
        .max_connections(5)
        .connect(&database_url)
        .await?;
    let pool = Arc::new(pool);

    // S3 Service 생성
    let object_storage = ObjectStorageServiceFactory::create(
        "s3",
        &s3_bucket,
        &s3_region,
        "",  // endpoint (S3는 불필요)
        &s3_access_key,
        &s3_secret_key,
    )
    .await?;
    let object_storage = Arc::new(object_storage);

    // Repository 생성
    let gc_repository = Arc::new(GcRepositoryImpl::new(pool.clone()));
    let gc_log_repository = Arc::new(GcLogRepositoryImpl::new(pool.clone()));

    // Service 생성
    let gc_service = GcServiceImpl::new(
        gc_repository,
        gc_log_repository,
        object_storage,
    );

    // 명령 실행
    match cli.command {
        Commands::TimeoutPending { grace_days, batch_size, dry_run } => {
            println!("🔄 Running Job A: Timeout Pending Snapshots");
            println!("   Grace Days: {}", grace_days);
            println!("   Batch Size: {}", batch_size);
            println!("   Dry-run: {}", dry_run);

            let results = gc_service
                .timeout_pending_snapshots(grace_days, batch_size, dry_run)
                .await?;

            let success_count = results.iter().filter(|r| r.success).count();
            let failed_count = results.len() - success_count;

            println!("✅ Job A completed:");
            println!("   Total: {}", results.len());
            println!("   Success: {}", success_count);
            println!("   Failed: {}", failed_count);
        }

        Commands::CleanupFailed { grace_days, batch_size, dry_run } => {
            println!("🗑️  Running Job B: Cleanup Failed Snapshots");
            println!("   Grace Days: {}", grace_days);
            println!("   Batch Size: {}", batch_size);
            println!("   Dry-run: {}", dry_run);

            let results = gc_service
                .cleanup_failed_snapshots(grace_days, batch_size, dry_run)
                .await?;

            let success_count = results.iter().filter(|r| r.success).count();
            let failed_count = results.len() - success_count;

            println!("✅ Job B completed:");
            println!("   Total: {}", results.len());
            println!("   Success: {}", success_count);
            println!("   Failed: {}", failed_count);
        }
    }

    Ok(())
}
```

| **6** | 전체 워크플로우 | 5일 전 PENDING + S3 | Job A → Job B | PENDING → FAILED → S3 삭제 |

#### 6.2 Python E2E 테스트 스크립트

**경로**: `pacs-server/tests/e2e/test_gc_e2e.py`

```python
#!/usr/bin/env python3
"""
GC Batch Job E2E 테스트

시나리오:
1. DB에 테스트 데이터 삽입
2. S3에 테스트 파일 업로드
3. GC Job 실행
4. 결과 검증
"""

import os
import sys
import time
import subprocess
import psycopg2
from datetime import datetime, timedelta
from minio import Minio
from minio.error import S3Error

# 환경 변수
DB_HOST = os.getenv("DB_HOST", "localhost")
DB_PORT = os.getenv("DB_PORT", "5433")
DB_NAME = os.getenv("DB_NAME", "pacs_test_db")
DB_USER = os.getenv("DB_USER", "pacs_user")
DB_PASSWORD = os.getenv("DB_PASSWORD", "pacs_password")

S3_ENDPOINT = os.getenv("S3_ENDPOINT", "localhost:9000")
S3_ACCESS_KEY = os.getenv("S3_ACCESS_KEY", "minioadmin")
S3_SECRET_KEY = os.getenv("S3_SECRET_KEY", "minioadmin")
S3_BUCKET = os.getenv("S3_BUCKET", "pacs-test-bucket")

GC_RUNNER_PATH = os.getenv("GC_RUNNER_PATH", "./target/debug/gc_runner")


class E2ETestRunner:
    def __init__(self):
        self.db_conn = None
        self.s3_client = None
        self.test_results = []

    def setup(self):
        """테스트 환경 초기화"""
        print("🔧 Setting up test environment...")

        # DB 연결
        self.db_conn = psycopg2.connect(
            host=DB_HOST,
            port=DB_PORT,
            dbname=DB_NAME,
            user=DB_USER,
            password=DB_PASSWORD
        )
        print("✅ Database connected")

        # S3 클라이언트 초기화
        self.s3_client = Minio(
            S3_ENDPOINT,
            access_key=S3_ACCESS_KEY,
            secret_key=S3_SECRET_KEY,
            secure=False
        )

        # 버킷 생성
        if not self.s3_client.bucket_exists(S3_BUCKET):
            self.s3_client.make_bucket(S3_BUCKET)
        print("✅ S3 bucket ready")

    def teardown(self):
        """테스트 환경 정리"""
        print("🧹 Cleaning up test environment...")

        if self.db_conn:
            self.db_conn.close()

        print("✅ Cleanup completed")

    def insert_test_annotation(self, snapshot_status, days_ago, snapshot_key):
        """테스트 어노테이션 삽입"""
        cursor = self.db_conn.cursor()
        updated_at = datetime.now() - timedelta(days=days_ago)

        cursor.execute(
            """
            INSERT INTO annotation_annotation (
                project_id, user_id, study_uid, tool_name, data, is_shared,
                snapshot_status, snapshot_image_key, updated_at
            )
            VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s)
            RETURNING id
            """,
            (1, 1, f"test-study-{snapshot_key}", "test-tool", "{}", False,
             snapshot_status, snapshot_key, updated_at)
        )

        annotation_id = cursor.fetchone()[0]
        self.db_conn.commit()
        cursor.close()

        return annotation_id

    def upload_test_file(self, file_key, content=b"test content"):
        """테스트 파일 S3 업로드"""
        from io import BytesIO

        self.s3_client.put_object(
            S3_BUCKET,
            file_key,
            BytesIO(content),
            len(content)
        )

    def file_exists_in_s3(self, file_key):
        """S3 파일 존재 확인"""
        try:
            self.s3_client.stat_object(S3_BUCKET, file_key)
            return True
        except S3Error:
            return False

    def get_annotation_status(self, annotation_id):
        """어노테이션 상태 조회"""
        cursor = self.db_conn.cursor()
        cursor.execute(
            "SELECT snapshot_status FROM annotation_annotation WHERE id = %s",
            (annotation_id,)
        )
        result = cursor.fetchone()
        cursor.close()
        return result[0] if result else None

    def get_gc_log_count(self, annotation_id):
        """GC 로그 개수 조회"""
        cursor = self.db_conn.cursor()
        cursor.execute(
            "SELECT COUNT(*) FROM gc_deletion_log WHERE annotation_id = %s",
            (annotation_id,)
        )
        count = cursor.fetchone()[0]
        cursor.close()
        return count

    def run_gc_job(self, job_type, grace_days, dry_run=False):
        """GC Job 실행"""
        cmd = [
            GC_RUNNER_PATH,
            job_type,
            f"--grace-days={grace_days}",
            "--batch-size=100",
            f"--dry-run={str(dry_run).lower()}"
        ]

        env = os.environ.copy()
        env["DATABASE_URL"] = f"postgresql://{DB_USER}:{DB_PASSWORD}@{DB_HOST}:{DB_PORT}/{DB_NAME}"
        env["S3_ENDPOINT"] = f"http://{S3_ENDPOINT}"
        env["S3_BUCKET"] = S3_BUCKET
        env["S3_ACCESS_KEY"] = S3_ACCESS_KEY
        env["S3_SECRET_KEY"] = S3_SECRET_KEY
        env["S3_REGION"] = "us-east-1"

        result = subprocess.run(cmd, env=env, capture_output=True, text=True)

        return {
            "returncode": result.returncode,
            "stdout": result.stdout,
            "stderr": result.stderr
        }

    def test_scenario_1_timeout_pending_dry_run(self):
        """
        시나리오 1: PENDING 타임아웃 (Dry-run)

        Given: 5일 전 PENDING 상태 어노테이션
        When: timeout-pending Job 실행 (grace_days=3, dry_run=true)
        Then:
          - DB 상태 변경 없음
          - GC 로그 기록됨 (dry_run=true)
        """
        print("\n📋 Test Scenario 1: PENDING Timeout (Dry-run)")

        # Given
        annotation_id = self.insert_test_annotation("pending", 5, "test-key-1")
        print(f"  ✅ Created annotation {annotation_id} (PENDING, 5 days ago)")

        # When
        result = self.run_gc_job("timeout-pending", 3, dry_run=True)
        assert result["returncode"] == 0, f"GC job failed: {result['stderr']}"
        print(f"  ✅ GC job executed successfully")

        # Then
        status = self.get_annotation_status(annotation_id)
        assert status == "pending", f"Status should remain 'pending', got '{status}'"
        print(f"  ✅ Status unchanged: {status}")

        log_count = self.get_gc_log_count(annotation_id)
        assert log_count == 1, f"Expected 1 log entry, got {log_count}"
        print(f"  ✅ GC log recorded: {log_count} entry")

        self.test_results.append(("Scenario 1", "PASSED"))

    def test_scenario_2_timeout_pending_actual_run(self):
        """
        시나리오 2: PENDING 타임아웃 (실제 실행)

        Given: 5일 전 PENDING 상태 어노테이션
        When: timeout-pending Job 실행 (grace_days=3, dry_run=false)
        Then:
          - DB 상태 PENDING → FAILED
          - GC 로그 기록됨 (dry_run=false)
        """
        print("\n📋 Test Scenario 2: PENDING Timeout (Actual Run)")

        # Given
        annotation_id = self.insert_test_annotation("pending", 5, "test-key-2")
        print(f"  ✅ Created annotation {annotation_id} (PENDING, 5 days ago)")

        # When
        result = self.run_gc_job("timeout-pending", 3, dry_run=False)
        assert result["returncode"] == 0, f"GC job failed: {result['stderr']}"
        print(f"  ✅ GC job executed successfully")

        # Then
        status = self.get_annotation_status(annotation_id)
        assert status == "failed", f"Status should be 'failed', got '{status}'"
        print(f"  ✅ Status changed: PENDING → {status}")

        log_count = self.get_gc_log_count(annotation_id)
        assert log_count == 1, f"Expected 1 log entry, got {log_count}"
        print(f"  ✅ GC log recorded: {log_count} entry")

        self.test_results.append(("Scenario 2", "PASSED"))

    def test_scenario_3_cleanup_failed_dry_run(self):

        시나리오 3: FAILED 정리 (Dry-run)

        Given: 10일 전 FAILED 상태 어노테이션 + S3 파일

        # Then

        Then:
          - 상태: PENDING → FAILED
          - S3 파일 삭제됨

        assert log_count == 2, f"Expected 2 log entries, got {log_count}"
        print(f"  ✅ GC logs recorded: {log_count} entries (Job A + Job B)")

        self.test_results.append(("Scenario 6", "PASSED"))

    def run_all_tests(self):

            self.test_scenario_2_timeout_pending_actual_run()
```

### 7.2 Kubernetes CronJob YAML

**경로**: `k8s/cronjobs/gc-job-a-timeout-pending.yaml`

```yaml
apiVersion: batch/v1
kind: CronJob
metadata:
  name: gc-job-a-timeout-pending
  namespace: pacs-system
spec:
  # 매일 새벽 2시 실행 (KST 기준 새벽 2시 = UTC 17시)
  schedule: "0 17 * * *"

  # 동시 실행 방지
  concurrencyPolicy: Forbid

  # 성공한 Job 1개만 보관
  successfulJobsHistoryLimit: 1

  # 실패한 Job 3개 보관
  failedJobsHistoryLimit: 3

  jobTemplate:
    spec:
      # Job 타임아웃 (1시간)
      activeDeadlineSeconds: 3600

      template:
        spec:
          restartPolicy: OnFailure

          containers:
          - name: gc-runner
            image: your-registry/pacs-server:latest
            imagePullPolicy: Always

            command: ["/usr/local/bin/gc_runner"]
            args:
              - "timeout-pending"
              - "--grace-days=3"
              - "--batch-size=1000"
              - "--dry-run=false"

            env:
            - name: DATABASE_URL
              valueFrom:
                secretKeyRef:
                  name: pacs-db-secret
                  key: database-url

            - name: S3_BUCKET
              valueFrom:
                configMapKeyRef:
                  name: pacs-config
                  key: s3-bucket

            - name: S3_REGION
              valueFrom:
                configMapKeyRef:
                  name: pacs-config
                  key: s3-region

            - name: S3_ACCESS_KEY
              valueFrom:
                secretKeyRef:
                  name: s3-secret
                  key: access-key

            - name: S3_SECRET_KEY
              valueFrom:
                secretKeyRef:
                  name: s3-secret
                  key: secret-key

            resources:
              requests:
                memory: "256Mi"
                cpu: "100m"
              limits:
                memory: "512Mi"
                cpu: "500m"
```

**경로**: `k8s/cronjobs/gc-job-b-cleanup-failed.yaml`

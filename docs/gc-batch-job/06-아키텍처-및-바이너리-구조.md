## 📋 아키텍처 결정

### 선택: 별도 바이너리 방식 ✅

**이유**:
1. **Rust 생태계 표준** - 대부분의 Rust 프로젝트가 멀티 바이너리 사용
2. **리소스 효율** - 각 바이너리가 필요한 의존성만 포함
3. **운영 유연성** - 독립적인 배포 및 스케일링
4. **간단한 구현** - 복잡한 모드 분기 불필요
5. **코드 공유** - 공통 로직은 `lib.rs`로 재사용

## 🔧 공통 라이브러리 (src/lib.rs)

```rust
// src/lib.rs
pub mod application;
pub mod infrastructure;
pub mod domain;

// 공통 설정 모듈
pub mod config {
    use std::env;
    
    #[derive(Debug, Clone)]
    pub struct DatabaseConfig {
        pub url: String,
        pub max_connections: u32,
    }
    
    impl DatabaseConfig {
        pub fn from_env() -> Self {
            Self {
                url: env::var("DATABASE_URL")
                    .expect("DATABASE_URL must be set"),
                max_connections: env::var("DB_MAX_CONNECTIONS")
                    .unwrap_or_else(|_| "10".to_string())
                    .parse()
                    .unwrap_or(10),
            }
        }
    }
    
    #[derive(Debug, Clone)]
    pub struct S3Config {
        pub bucket_name: String,
        pub region: String,
    }
    
    impl S3Config {
        pub fn from_env() -> Self {
            Self {
                bucket_name: env::var("S3_BUCKET_NAME")
                    .expect("S3_BUCKET_NAME must be set"),
                region: env::var("AWS_REGION")
                    .unwrap_or_else(|_| "ap-northeast-2".to_string()),
            }
        }
    }
    
    #[derive(Debug, Clone)]
    pub struct GcConfig {
        pub dry_run: bool,
        pub batch_size: i32,
        pub pending_grace_days: i32,
        pub failed_grace_days: i32,
    }
    
    impl GcConfig {
        pub fn from_env() -> Self {
            Self {
                dry_run: env::var("GC_DRY_RUN")
                    .unwrap_or_else(|_| "true".to_string())
                    .parse()
                    .unwrap_or(true),
                batch_size: env::var("GC_BATCH_SIZE")
                    .unwrap_or_else(|_| "1000".to_string())
                    .parse()
                    .unwrap_or(1000),
                pending_grace_days: env::var("GC_PENDING_GRACE_DAYS")
                    .unwrap_or_else(|_| "3".to_string())
                    .parse()
                    .unwrap_or(3),
                failed_grace_days: env::var("GC_FAILED_GRACE_DAYS")
                    .unwrap_or_else(|_| "7".to_string())
                    .parse()
                    .unwrap_or(7),
            }
        }
    }
}

// 공통 에러 타입
pub mod error {
    use thiserror::Error;
    
    #[derive(Error, Debug)]
    pub enum ServiceError {
        #[error("Database error: {0}")]
        DatabaseError(#[from] sqlx::Error),
        
        #[error("S3 error: {0}")]
        S3Error(String),
        
        #[error("Not found: {0}")]
        NotFound(String),
        
        #[error("Unauthorized: {0}")]
        Unauthorized(String),
        
        #[error("External service error: {0}")]
        ExternalServiceError(String),
    }
}
```

## 🚀 API 서버 바이너리 (src/bin/server.rs)

```rust
// src/bin/server.rs
use pacs_server::config::{DatabaseConfig, S3Config};
use axum::{Router, routing::get};
use std::net::SocketAddr;

#[tokio::main]
async fn main() -> Result<(), Box<dyn std::error::Error>> {
    // 로깅 초기화
    tracing_subscriber::fmt()
        .with_env_filter(
            tracing_subscriber::EnvFilter::from_default_env()
                .add_directive(tracing::Level::INFO.into())
        )
        .init();
    
    tracing::info!("Starting PACS API Server...");
    
    // 설정 로드
    let db_config = DatabaseConfig::from_env();
    let s3_config = S3Config::from_env();
    
    // DB 연결 풀
    let pool = sqlx::postgres::PgPoolOptions::new()
        .max_connections(db_config.max_connections)
        .connect(&db_config.url)
        .await?;
    
    tracing::info!("Database connected: {} connections", db_config.max_connections);
    
    // 라우터 설정
    let app = Router::new()
        .route("/health", get(health_check));
    
    // 서버 시작
    let addr = SocketAddr::from(([0, 0, 0, 0], 8080));
    let listener = tokio::net::TcpListener::bind(addr).await?;
    
    tracing::info!("API Server listening on {}", addr);
    
    axum::serve(listener, app).await?;
    
    Ok(())
}

async fn health_check() -> &'static str {
    "OK"
}
```

## 🔄 GC 배치 바이너리 (src/bin/gc_runner.rs)

```rust
// src/bin/gc_runner.rs
use pacs_server::application::services::gc_service::GcService;
use pacs_server::infrastructure::repositories::gc_repository::GcRepository;
use pacs_server::config::{DatabaseConfig, S3Config, GcConfig};
use clap::Parser;

#[derive(Parser, Debug)]
#[command(name = "gc_runner")]
#[command(about = "PACS GC Batch Job Runner", version)]
struct Cli {
    /// Job type: snapshot, mask, orphan, all
    #[arg(long, default_value = "snapshot")]
    job: String,
    
    /// Dry-run mode (no actual deletion)
    #[arg(long)]
    dry_run: Option<bool>,
    
    /// Batch size (overrides env var)
    #[arg(long)]
    batch_size: Option<i32>,
}

#[tokio::main]
async fn main() -> Result<(), Box<dyn std::error::Error>> {
    // 로깅 초기화
    tracing_subscriber::fmt()
        .with_env_filter(
            tracing_subscriber::EnvFilter::from_default_env()
                .add_directive(tracing::Level::INFO.into())
        )
        .json()  // JSON 로그 (Kubernetes 로그 수집용)
        .init();
    
    let cli = Cli::parse();
    
    tracing::info!(
        job = %cli.job,
        dry_run = ?cli.dry_run,
        batch_size = ?cli.batch_size,
        "Starting GC Batch Job"
    );
    
    // 설정 로드
    let db_config = DatabaseConfig::from_env();
    let s3_config = S3Config::from_env();
    let mut gc_config = GcConfig::from_env();
    
    // CLI 인자로 오버라이드
    if let Some(dry_run) = cli.dry_run {
        gc_config.dry_run = dry_run;
    }
    if let Some(batch_size) = cli.batch_size {
        gc_config.batch_size = batch_size;
    }
    
    tracing::info!("GC Config: {:?}", gc_config);
    
    // DB 연결 (배치는 적은 연결로 충분)
    let pool = sqlx::postgres::PgPoolOptions::new()
        .max_connections(5)
        .connect(&db_config.url)
        .await?;
    
    tracing::info!("Database connected");
    
    // S3 클라이언트
    let aws_config = aws_config::from_env()
        .region(aws_sdk_s3::config::Region::new(s3_config.region.clone()))
        .load()
        .await;
    let s3_client = aws_sdk_s3::Client::new(&aws_config);
    
    tracing::info!("S3 client initialized: bucket={}", s3_config.bucket_name);
    
    // GC 서비스 초기화
    let gc_repository = GcRepository::new(pool.clone());
    let gc_service = GcService::new(
        gc_repository,
        s3_client,
        s3_config.bucket_name,
    );
    
    // Job 실행
    let result = match cli.job.as_str() {
        "snapshot" => run_snapshot_gc(&gc_service, &gc_config).await,
        "mask" => run_mask_gc(&gc_service, &gc_config).await,
        "orphan" => run_orphan_gc(&gc_service, &gc_config).await,
        "all" => run_all_gc(&gc_service, &gc_config).await,
        _ => {
            tracing::error!("Unknown job type: {}", cli.job);
            Err("Unknown job type".into())
        }
    };
    
    match result {
        Ok(_) => {
            tracing::info!("GC Batch Job completed successfully");
            Ok(())
        }
        Err(e) => {
            tracing::error!("GC Batch Job failed: {}", e);
            Err(e)
        }
    }
}

async fn run_snapshot_gc(
    gc_service: &GcService,
    config: &GcConfig,
) -> Result<(), Box<dyn std::error::Error>> {
    tracing::info!("Running Snapshot GC...");
    
    // PENDING 타임아웃
    let pending_results = gc_service
        .timeout_pending_snapshots(
            config.pending_grace_days,
            config.batch_size,
            config.dry_run,
        )
        .await?;
    
    tracing::info!(
        count = pending_results.len(),
        "PENDING timeout completed"
    );
    
    // FAILED 정리
    let failed_results = gc_service
        .cleanup_failed_snapshots(
            config.failed_grace_days,
            config.batch_size,
            config.dry_run,
        )
        .await?;
    
    tracing::info!(
        count = failed_results.len(),
        "FAILED cleanup completed"
    );
    
    Ok(())
}

async fn run_mask_gc(
    _gc_service: &GcService,
    _config: &GcConfig,
) -> Result<(), Box<dyn std::error::Error>> {
    tracing::info!("Running Mask GC...");
    // TODO: Mask GC 구현
    Ok(())
}

async fn run_orphan_gc(
    _gc_service: &GcService,
    _config: &GcConfig,
) -> Result<(), Box<dyn std::error::Error>> {
    tracing::info!("Running Orphan GC...");
    // TODO: Orphan GC 구현

async fn run_all_gc(
    gc_service: &GcService,
    config: &GcConfig,
) -> Result<(), Box<dyn std::error::Error>> {
    run_snapshot_gc(gc_service, config).await?;
    run_mask_gc(gc_service, config).await?;
    run_orphan_gc(gc_service, config).await?;
```

## 🎯 장점 요약

1. **리소스 효율**
   - API 서버: Axum, Tower 등 포함
   - GC 배치: Clap만 포함 (더 작은 바이너리)

2. **독립 배포**
   - GC 로직 수정 시 API 서버 재시작 불필요
   - 각 바이너리 독립적으로 버전 관리

3. **코드 공유**
   - 공통 로직은 `lib.rs`로 재사용
   - 중복 코드 없음

4. **운영 유연성**
   - API 서버와 GC 배치를 독립적으로 스케일
   - 배치 작업 실패 시 API 서버에 영향 없음
